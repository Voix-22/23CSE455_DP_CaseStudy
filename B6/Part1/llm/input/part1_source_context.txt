     1	using System.Linq.Expressions;
     2	using FlemanApi.Data;
     3	using Microsoft.EntityFrameworkCore;
     4	
     5	namespace FlemanApi.Repository;
     6	
     7	// EF Core-backed implementation of IGenericRepository — requirement #8.
     8	// Reused as-is by every entity that doesn't need custom queries; entities
     9	// that do (e.g. Car, BookingHeader) get a thin sibling interface with just
    10	// the extra query methods, still built on this same DbContext.
    11	public class GenericRepository<TEntity, TKey> : IGenericRepository<TEntity, TKey>
    12	    where TEntity : class
    13	{
    14	    protected readonly FlemanDbContext Context;
    15	    protected readonly DbSet<TEntity> DbSet;
    16	
    17	    public GenericRepository(FlemanDbContext context)
    18	    {
    19	        Context = context;
    20	        DbSet = context.Set<TEntity>();
    21	    }
    22	
    23	    public async Task<TEntity?> GetByIdAsync(TKey id)
    24	    {
    25	        if (id is null) return null;
    26	        return await DbSet.FindAsync(id);
    27	    }
    28	
    29	    public async Task<IReadOnlyList<TEntity>> GetAllAsync() =>
    30	        await DbSet.AsNoTracking().ToListAsync();
    31	
    32	    public async Task<IReadOnlyList<TEntity>> FindAsync(Expression<Func<TEntity, bool>> predicate) =>
    33	        await DbSet.AsNoTracking().Where(predicate).ToListAsync();
    34	
    35	    public async Task<TEntity?> FirstOrDefaultAsync(Expression<Func<TEntity, bool>> predicate) =>
    36	        await DbSet.FirstOrDefaultAsync(predicate);
    37	
    38	    public async Task<bool> ExistsAsync(Expression<Func<TEntity, bool>> predicate) =>
    39	        await DbSet.AnyAsync(predicate);
    40	
    41	    public async Task<TEntity> AddAsync(TEntity entity)
    42	    {
    43	        await DbSet.AddAsync(entity);
    44	        return entity;
    45	    }
    46	
    47	    public void Update(TEntity entity) => DbSet.Update(entity);
    48	
    49	    public void Remove(TEntity entity) => DbSet.Remove(entity);
    50	
    51	    public Task SaveChangesAsync() => Context.SaveChangesAsync();
    52	}

     1	using FlemanApi.Data;
     2	using FlemanApi.Models;
     3	using Microsoft.EntityFrameworkCore;
     4	
     5	namespace FlemanApi.Repository;
     6	
     7	public class AvailabilityRepository : IAvailabilityRepository
     8	{
     9	    private static readonly BookingStatus[] ActiveStatuses = { BookingStatus.CONFIRMED, BookingStatus.ONGOING };
    10	
    11	    private readonly FlemanDbContext _context;
    12	
    13	    public AvailabilityRepository(FlemanDbContext context)
    14	    {
    15	        _context = context;
    16	    }
    17	
    18	    public Task<long> CountCapacityAsync(long hubId, long carTypeId) =>
    19	        _context.Cars.LongCountAsync(c =>
    20	            c.HubId == hubId && c.CarTypeId == carTypeId && c.Status != CarStatus.UNDER_MAINTENANCE);
    21	
    22	    public Task<long> CountOverlappingAsync(
    23	        long hubId, long carTypeId, DateTime pickupDatetime, DateTime returnDatetime, long? excludeBookingId) =>
    24	        _context.BookingHeaders.LongCountAsync(b =>
    25	            b.PickupHubId == hubId
    26	            && b.CarTypeId == carTypeId
    27	            && b.BookingStatus.HasValue && ActiveStatuses.Contains(b.BookingStatus.Value)
    28	            && b.PickupDatetime < returnDatetime
    29	            && b.ReturnDatetime > pickupDatetime
    30	            && (excludeBookingId == null || b.BookingId != excludeBookingId.Value));
    31	
    32	    public async Task<IReadOnlyList<CarTypeCountProjection>> CountByCarTypeAsync(long? hubId)
    33	    {
    34	        var query = _context.Cars.AsNoTracking().AsQueryable();
    35	        if (hubId is not null) query = query.Where(c => c.HubId == hubId.Value);
    36	
    37	        var grouped = await query
    38	            .GroupBy(c => c.CarTypeId)
    39	            .Select(g => new CarTypeCountProjection(
    40	                g.Key,
    41	                g.LongCount(),
    42	                g.LongCount(c => c.Status == CarStatus.AVAILABLE),
    43	                g.LongCount(c => c.Status == CarStatus.BOOKED),
    44	                g.LongCount(c => c.Status == CarStatus.UNDER_MAINTENANCE)))
    45	            .ToListAsync();
    46	
    47	        return grouped;
    48	    }
    49	}

     1	using AutoMapper;
     2	using FlemanApi.DTO;
     3	using FlemanApi.Exceptions;
     4	using FlemanApi.Models;
     5	using FlemanApi.Repository;
     6	
     7	namespace FlemanApi.Service;
     8	
     9	// Mirrors com.fleman.service.impl.VehicleServiceImpl.
    10	public class VehicleService : IVehicleService
    11	{
    12	    private readonly IGenericRepository<Car, long> _cars;
    13	    private readonly IGenericRepository<CarType, long> _carTypes;
    14	    private readonly IAvailabilityRepository _availability;
    15	    private readonly IMapper _mapper;
    16	
    17	    public VehicleService(
    18	        IGenericRepository<Car, long> cars,
    19	        IGenericRepository<CarType, long> carTypes,
    20	        IAvailabilityRepository availability,
    21	        IMapper mapper)
    22	    {
    23	        _cars = cars;
    24	        _carTypes = carTypes;
    25	        _availability = availability;
    26	        _mapper = mapper;
    27	    }
    28	
    29	    public async Task<IReadOnlyList<CarTypeDTO>> GetCarTypesAsync() =>
    30	        _mapper.Map<List<CarTypeDTO>>(await _carTypes.GetAllAsync());
    31	
    32	    public async Task<IReadOnlyList<CarTypeDTO>> GetCarTypesForHubAsync(
    33	        long? hubId, DateTime? pickupDatetime, DateTime? returnDatetime)
    34	    {
    35	        if (hubId is null) return await GetCarTypesAsync();
    36	
    37	        var carTypes = await _carTypes.GetAllAsync();
    38	        var result = new List<CarTypeDTO>();
    39	
    40	        // Category-level (not per-car) availability: capacity is the
    41	        // physical fleet of that type at this hub minus anything under
    42	        // maintenance; demand is how many CONFIRMED/ONGOING bookings
    43	        // already reserve that type here for an overlapping date range.
    44	        // Every type is always returned — availableCount 0 is shown
    45	        // disabled on the vehicle-selection page, not hidden.
    46	        foreach (var ct in carTypes)
    47	        {
    48	            var dto = _mapper.Map<CarTypeDTO>(ct);
    49	            var capacity = await _availability.CountCapacityAsync(hubId.Value, ct.CarTypeId);
    50	            var overlapping = pickupDatetime is not null && returnDatetime is not null
    51	                ? await _availability.CountOverlappingAsync(hubId.Value, ct.CarTypeId, pickupDatetime.Value, returnDatetime.Value, null)
    52	                : 0;
    53	            dto.AvailableCount = (int)Math.Max(0, capacity - overlapping);
    54	            result.Add(dto);
    55	        }
    56	
    57	        return result;
    58	    }
    59	
    60	    public async Task<IReadOnlyList<CarDTO>> GetAvailableCarsAsync(
    61	        long? hubId, DateTime? pickupDatetime, DateTime? returnDatetime)
    62	    {
    63	        if (hubId is null)
    64	        {
    65	            var all = await _cars.GetAllAsync();
    66	            var dtos = new List<CarDTO>();
    67	            foreach (var c in all) dtos.Add(await ToDtoAsync(c, true));
    68	            return dtos;
    69	        }
    70	
    71	        // Two different questions, deliberately not conflated: "which cars
    72	        // exist at this hub" (every one of them) vs "which of them are
    73	        // physically free right now" (plain status check — individual
    74	        // cars aren't reserved by date, only the category is).
    75	        var atHub = await _cars.FindAsync(c => c.HubId == hubId.Value);
    76	        var result = new List<CarDTO>();
    77	        foreach (var c in atHub) result.Add(await ToDtoAsync(c, c.Status == CarStatus.AVAILABLE));
    78	        return result;
    79	    }
    80	
    81	    public async Task<CarDTO> GetCarByIdAsync(long carId)
    82	    {
    83	        var car = await _cars.GetByIdAsync(carId)
    84	            ?? throw new ResourceNotFoundException($"Car not found: {carId}");
    85	        return await ToDtoAsync(car, car.Status == CarStatus.AVAILABLE);
    86	    }
    87	
    88	    public async Task<CarTypeDTO> GetCarTypeByIdAsync(long carTypeId)
    89	    {
    90	        var carType = await _carTypes.GetByIdAsync(carTypeId)
    91	            ?? throw new ResourceNotFoundException($"Car type not found: {carTypeId}");
    92	        return _mapper.Map<CarTypeDTO>(carType);
    93	    }
    94	
    95	    private async Task<CarDTO> ToDtoAsync(Car car, bool bookableNow)
    96	    {
    97	        var dto = _mapper.Map<CarDTO>(car);
    98	        dto.BookableNow = bookableNow;
    99	        // No relationship — an explicit second lookup by the plain
   100	        // CarTypeId column to build the nested CarTypeDTO the frontend expects.
   101	        var carType = await _carTypes.GetByIdAsync(car.CarTypeId);
   102	        if (carType is not null) dto.CarType = _mapper.Map<CarTypeDTO>(carType);
   103	        return dto;
   104	    }
   105	}

    78	    public async Task<BookingResponseDTO> CreateBookingAsync(CreateBookingRequest request)
    79	    {
    80	        _logger.LogInformation("Booking process started");
    81	
    82	        long customerId;
    83	        if (_currentUser.IsAuthenticated)
    84	        {
    85	            if (_currentUser.GetCurrentRole() == "STAFF")
    86	            {
    87	                if (request.CustomerId is null) throw new ApiException("Customer is required.", HttpStatusCode.BadRequest);
    88	                customerId = request.CustomerId.Value;
    89	            }
    90	            else
    91	            {
    92	                customerId = _currentUser.GetCurrentUserId();
    93	            }
    94	        }
    95	        else
    96	        {
    97	            if (request.CustomerId is null) throw new ApiException("Customer is required.", HttpStatusCode.BadRequest);
    98	            customerId = request.CustomerId.Value;
    99	        }
   100	
   101	        var carType = await _carTypes.GetByIdAsync(request.CarTypeId!.Value)
   102	            ?? throw new ResourceNotFoundException($"Car type not found: {request.CarTypeId}");
   103	
   104	        if (!(request.ReturnDatetime!.Value > request.PickupDatetime!.Value))
   105	            throw new ApiException("Return date must be after pickup date", HttpStatusCode.BadRequest);
   106	
   107	        // Faithfully mirrors the Java source: this checks request.CustomerId
   108	        // (the raw request field), not the resolved customerId above — the
   109	        // two can differ when a logged-in customer's JWT id doesn't match
   110	        // whatever customerId they happened to submit in the body.
   111	        if (!await _customers.ExistsAsync(c => c.CustomerId == request.CustomerId))
   112	            throw new ResourceNotFoundException($"Customer not found: {request.CustomerId}");
   113	
   114	        var dropHubId = request.DropHubId ?? request.PickupHubId!.Value;
   115	        if (!await _hubs.ExistsAsync(h => h.HubId == request.PickupHubId))
   116	            throw new ResourceNotFoundException($"Hub not found: {request.PickupHubId}");
   117	        if (!await _hubs.ExistsAsync(h => h.HubId == dropHubId))
   118	            throw new ResourceNotFoundException($"Hub not found: {dropHubId}");
   119	
   120	        await AssertCategoryHasCapacityAsync(request.PickupHubId!.Value, request.CarTypeId!.Value,
   121	            request.PickupDatetime.Value, request.ReturnDatetime.Value, null);
   122	
   123	        var header = new BookingHeader
   124	        {
   125	            ConfirmationNo = await GenerateConfirmationNoAsync(),
   126	            BookingDate = DateTime.Now,
   127	            CustomerId = customerId,
   128	            CarTypeId = carType.CarTypeId,
   129	            CarId = null,
   130	            PickupHubId = request.PickupHubId.Value,
   131	            DropHubId = dropHubId,
   132	            PickupDatetime = request.PickupDatetime.Value,
   133	            ReturnDatetime = request.ReturnDatetime.Value,
   134	            BookingStatus = BookingStatus.CONFIRMED,
   135	            Remarks = request.Remarks ?? string.Empty,
   136	        };
   137	
   138	        var days = DaysBetween(request.PickupDatetime.Value, request.ReturnDatetime.Value);
   139	        var rentalAmount = RentalRateCalculator.CalculateRentalAmount(carType, days);
   140	
   141	        await _bookings.AddAsync(header);
   142	        await _bookings.SaveChangesAsync();
   143	
   144	        var addonAmount = 0.0;
   145	        if (request.Addons is not null)
   146	        {
   147	            foreach (var line in request.Addons)
   148	            {
   149	                var addon = line.AddonId is null ? null : await _addons.GetByIdAsync(line.AddonId.Value);
   150	                if (addon is null) continue;
   151	                var subtotal = addon.DailyRate * (line.Quantity ?? 0) * days;
   152	                await _bookingDetails.AddAsync(new BookingDetail
   153	                {
   154	                    BookingId = header.BookingId,
   155	                    AddonId = addon.AddonId,
   156	                    AddonRate = addon.DailyRate,
   157	                    Quantity = line.Quantity,
   158	                    Subtotal = subtotal,
   159	                });
   160	                addonAmount += subtotal;
   161	            }
   162	        }
   163	
   164	        header.EstimatedAmount = rentalAmount + addonAmount;
   165	        await _bookings.SaveChangesAsync();
   166	
   167	        try
   168	        {
   169	            var customer = await _customers.GetByIdAsync(header.CustomerId);
   170	            if (customer is not null)
   171	            {
   172	                var body = $"""
   173	                    Dear {customer.FullName},
   174	
   175	                    Your booking has been confirmed successfully.
   176	
   177	                    Booking Details
   178	                    -------------------------
   179	                    Booking No : {header.ConfirmationNo}
   180	                    Vehicle Type : {carType.CarTypeName}
   181	                    Pickup Date : {header.PickupDatetime}
   182	                    Return Date : {header.ReturnDatetime}
   183	                    Total Amount : ₹{header.EstimatedAmount:F2}
   184	
   185	                    Thank you for choosing Fleeman.
   186	
   187	                    Regards,
   188	                    Fleeman Team
   189	                    """;
   190	                await _emailService.SendEmailAsync(customer.Email, "Booking Confirmation - Fleeman", body);
   191	            }
   192	        }
   193	        catch (Exception ex)
   194	        {
   195	            _logger.LogWarning(ex, "Failed to send booking confirmation email for {ConfirmationNo}", header.ConfirmationNo);
   196	        }
   197	
   198	        _logger.LogInformation("Booking process completed, confirmationNo={ConfirmationNo}", header.ConfirmationNo);
   199	        return await ToResponseDtoAsync(header);
   200	    }

   223	    public async Task<BookingResponseDTO> ModifyBookingAsync(string confirmationNo, ModifyBookingRequest request)
   224	    {
   225	        var header = await FindHeaderAsync(confirmationNo);
   226	        ValidateBookingAccess(header);
   227	
   228	        // Once a real car has been handed to the customer, the pickup leg
   229	        // is history — only the still-pending return leg may still change.
   230	        var pickupHubChanged = request.PickupHubId is not null && request.PickupHubId != header.PickupHubId;
   231	        var pickupTimeChanged = request.PickupDatetime is not null && request.PickupDatetime != header.PickupDatetime;
   232	        if ((pickupHubChanged || pickupTimeChanged) && header.CarId is not null)
   233	        {
   234	            throw new ApiException("Vehicle has already been handed over — pickup location and time can no longer be changed", HttpStatusCode.Conflict);
   235	        }
   236	
   237	        if (request.PickupHubId is not null)
   238	        {
   239	            if (!await _hubs.ExistsAsync(h => h.HubId == request.PickupHubId))
   240	                throw new ResourceNotFoundException($"Hub not found: {request.PickupHubId}");
   241	            header.PickupHubId = request.PickupHubId.Value;
   242	        }
   243	        if (request.DropHubId is not null)
   244	        {
   245	            if (!await _hubs.ExistsAsync(h => h.HubId == request.DropHubId))
   246	                throw new ResourceNotFoundException($"Hub not found: {request.DropHubId}");
   247	            header.DropHubId = request.DropHubId.Value;
   248	        }
   249	        if (request.PickupDatetime is not null) header.PickupDatetime = request.PickupDatetime.Value;
   250	        if (request.ReturnDatetime is not null) header.ReturnDatetime = request.ReturnDatetime.Value;
   251	        if (request.Remarks is not null) header.Remarks = request.Remarks;
   252	
   253	        // Changing the reserved category: only while no real car has been
   254	        // handed over yet.
   255	        var typeChanged = request.CarTypeId is not null && request.CarTypeId != header.CarTypeId;
   256	        if (typeChanged)
   257	        {
   258	            if (header.CarId is not null)
   259	                throw new ApiException("Vehicle has already been handed over — category can no longer be changed", HttpStatusCode.Conflict);
   260	            if (!await _carTypes.ExistsAsync(ct => ct.CarTypeId == request.CarTypeId))
   261	                throw new ResourceNotFoundException($"Car type not found: {request.CarTypeId}");
   262	            header.CarTypeId = request.CarTypeId!.Value;
   263	        }
   264	
   265	        // Any change to hub, dates, or category needs a fresh capacity
   266	        // check against the booking's own final values, excluding itself
   267	        // from the demand count. Skipped once a car is already assigned.
   268	        var needsRecheck = typeChanged || request.PickupHubId is not null
   269	            || request.PickupDatetime is not null || request.ReturnDatetime is not null;
   270	        if (needsRecheck && header.CarId is null)
   271	        {
   272	            await AssertCategoryHasCapacityAsync(header.PickupHubId, header.CarTypeId,
   273	                header.PickupDatetime, header.ReturnDatetime, header.BookingId);
   274	        }
   275	
   276	        // Replacing the add-on lines: null means "leave as-is", an empty
   277	        // list means "every add-on was removed". Locked once a car has
   278	        // been handed over, same as the pickup-hub/time guard above.
   279	        if (request.Addons is not null && header.CarId is not null)
   280	        {
   281	            throw new ApiException("Vehicle has already been handed over — add-ons can no longer be changed", HttpStatusCode.Conflict);
   282	        }
   283	
   284	        var days = DaysBetween(header.PickupDatetime, header.ReturnDatetime);
   285	        double addonAmount;
   286	        if (request.Addons is not null)
   287	        {
   288	            var existingLines = await _bookingDetails.FindAsync(d => d.BookingId == header.BookingId);
   289	            foreach (var line in existingLines) _bookingDetails.Remove(line);
   290	
   291	            var total = 0.0;
   292	            foreach (var line in request.Addons)
   293	            {
   294	                var addon = line.AddonId is null ? null : await _addons.GetByIdAsync(line.AddonId.Value);
   295	                if (addon is null) continue;
   296	                var subtotal = addon.DailyRate * (line.Quantity ?? 0) * days;
   297	                await _bookingDetails.AddAsync(new BookingDetail
   298	                {
   299	                    BookingId = header.BookingId,
   300	                    AddonId = addon.AddonId,
   301	                    AddonRate = addon.DailyRate,
   302	                    Quantity = line.Quantity,
   303	                    Subtotal = subtotal,
   304	                });
   305	                total += subtotal;
   306	            }
   307	            addonAmount = total;
   308	        }
   309	        else
   310	        {
   311	            var currentLines = await _bookingDetails.FindAsync(d => d.BookingId == header.BookingId);
   312	            addonAmount = currentLines.Sum(d => d.Subtotal ?? 0);
   313	        }
   314	
   315	        var currentCarType = await _carTypes.GetByIdAsync(header.CarTypeId);
   316	        var rentalAmount = currentCarType is not null ? RentalRateCalculator.CalculateRentalAmount(currentCarType, days) : 0.0;
   317	        header.EstimatedAmount = rentalAmount + addonAmount;
   318	
   319	        await _bookings.SaveChangesAsync();
   320	        return await ToResponseDtoAsync(header);
   321	    }

   323	    public async Task<BookingResponseDTO> CancelBookingAsync(string confirmationNo)
   324	    {
   325	        var header = await FindHeaderAsync(confirmationNo);
   326	        ValidateBookingAccess(header);
   327	        header.BookingStatus = BookingStatus.CANCELLED;
   328	
   329	        // A car has only ever been touched if it was actually handed over —
   330	        // release it back to the fleet in that case. Otherwise there's

   332	        if (header.CarId is not null)

   338	                car.IsAvailable = true;
   339	            }
   340	        }
   341	
   342	        await _bookings.SaveChangesAsync();
   343	        return await ToResponseDtoAsync(header);
   344	    }

   346	    // Capacity (physical fleet of this type at this hub, minus maintenance)
   347	    // minus demand (other CONFIRMED/ONGOING bookings whose dates overlap)
   348	    // must leave at least one car free, or the reservation is rejected.
   349	    private async Task AssertCategoryHasCapacityAsync(
   350	        long hubId, long carTypeId, DateTime pickupDatetime, DateTime returnDatetime, long? excludeBookingId)
   351	    {
   352	        var capacity = await _availability.CountCapacityAsync(hubId, carTypeId);
   353	        var overlapping = await _availability.CountOverlappingAsync(hubId, carTypeId, pickupDatetime, returnDatetime, excludeBookingId);
   354	        if (capacity - overlapping <= 0)
   355	        {
   356	            throw new ApiException("That car type is no longer available at this hub for these dates", HttpStatusCode.Conflict);
   357	        }
   358	    }

   373	    // Fully random each time (3 letters + 5 digits, e.g. "QXR48213") —
   374	    // re-rolls on the rare chance of a collision with an existing
   375	    // confirmation number.
   376	    private async Task<string> GenerateConfirmationNoAsync()
   377	    {
   378	        string candidate;
   379	        do
   380	        {
   381	            var chars = new char[ConfirmationLetterCount + ConfirmationDigitCount];
   382	            for (var i = 0; i < ConfirmationLetterCount; i++)
   383	                chars[i] = ConfirmationLetters[RandomNumberGenerator.GetInt32(ConfirmationLetters.Length)];
   384	            for (var i = 0; i < ConfirmationDigitCount; i++)
   385	                chars[ConfirmationLetterCount + i] = ConfirmationDigits[RandomNumberGenerator.GetInt32(ConfirmationDigits.Length)];
   386	            candidate = new string(chars);
   387	        } while (await _bookings.ExistsAsync(b => b.ConfirmationNo.ToLower() == candidate.ToLower()));
   388	        return candidate;
   389	    }

   391	    private static int DaysBetween(DateTime pickup, DateTime returnDt)
   392	    {
   393	        var hours = (returnDt - pickup).TotalHours;
   394	        var fullDays = (long)Math.Ceiling(hours / 24.0);
   395	        return (int)Math.Max(1, fullDays);
   396	    }

   453	            Subtotal = detail.Subtotal,
   454	            Addon = allAddons.FirstOrDefault(a => a.AddonId == detail.AddonId),
   455	        }).ToList();
   456	
   457	        return dto;
   458	    }

     1	using Microsoft.Extensions.Logging;
     2	using Microsoft.Extensions.Options;
     3	using MimeKit;
     4	using MailKit.Net.Smtp;
     5	using MailKit.Security;
     6	
     7	namespace FlemanApi.Service;
     8	
     9	public class MailKitEmailSender : IEmailSender
    10	{
    11	    private readonly MailSettings _settings;
    12	    private readonly ILogger<MailKitEmailSender> _logger;
    13	
    14	    public MailKitEmailSender(IOptions<MailSettings> settings, ILogger<MailKitEmailSender> logger)
    15	    {
    16	        _settings = settings.Value;
    17	        _logger = logger;
    18	    }
    19	
    20	    public async Task SendPlainAsync(string to, string subject, string body)
    21	    {
    22	        var message = BuildMessage(to, subject);
    23	        message.Body = new TextPart("plain") { Text = body };
    24	        await SendAsync(message);
    25	    }
    26	
    27	    public async Task SendWithAttachmentAsync(string to, string subject, string body, string attachmentName, byte[] attachment)
    28	    {
    29	        var message = BuildMessage(to, subject);
    30	        var builder = new BodyBuilder { TextBody = body };
    31	        builder.Attachments.Add(attachmentName, attachment);
    32	        message.Body = builder.ToMessageBody();
    33	        await SendAsync(message);
    34	    }
    35	
    36	    private MimeMessage BuildMessage(string to, string subject)
    37	    {
    38	        var message = new MimeMessage();
    39	        message.From.Add(MailboxAddress.Parse(string.IsNullOrWhiteSpace(_settings.Username) ? "no-reply@wandercar.local" : _settings.Username));
    40	        message.To.Add(MailboxAddress.Parse(to));
    41	        message.Subject = subject;
    42	        return message;
    43	    }
    44	
    45	    private async Task SendAsync(MimeMessage message)
    46	    {
    47	        if (string.IsNullOrWhiteSpace(_settings.Host) || string.IsNullOrWhiteSpace(_settings.Username))
    48	        {
    49	            // No SMTP credentials configured (e.g. fresh checkout, secrets
    50	            // not yet filled in) — log instead of throwing, matching the
    51	            // best-effort "swallow and continue" behaviour every caller in
    52	            // the Java app already wraps its email sends in.
    53	            _logger.LogInformation("Mail not configured — skipping send of {Subject} to {To}", message.Subject, message.To);
    54	            return;
    55	        }
    56	
    57	        using var client = new SmtpClient();
    58	        await client.ConnectAsync(_settings.Host, _settings.Port,
    59	            _settings.UseStartTls ? SecureSocketOptions.StartTls : SecureSocketOptions.Auto);
    60	        await client.AuthenticateAsync(_settings.Username, _settings.Password);
    61	        await client.SendAsync(message);
    62	        await client.DisconnectAsync(true);
    63	    }
    64	}

     3	using FlemanApi.Exceptions;

     5	namespace FlemanApi.Middleware;
     6	
     7	// Mirrors com.fleman.exception.GlobalExceptionHandler (@RestControllerAdvice) —
     8	// requirement #4. Every unhandled exception, anywhere in the pipeline, comes
     9	// out the wire as the same { message, status, timestamp } JSON shape.

    11	// .InvalidModelStateResponseFactory registered in Program.cs (Java's
    12	// MethodArgumentNotValidException equivalent), so this middleware only needs
    13	// to cover ApiException and everything else.
    14	public class ExceptionHandlingMiddleware
    15	{
    16	    private readonly RequestDelegate _next;
    17	    private readonly ILogger<ExceptionHandlingMiddleware> _logger;
    18	    private static readonly JsonSerializerOptions JsonOptions = new()
    19	    {
    20	        PropertyNamingPolicy = JsonNamingPolicy.CamelCase
    21	    };
    22	
    23	    public ExceptionHandlingMiddleware(RequestDelegate next, ILogger<ExceptionHandlingMiddleware> logger)
    24	    {
    25	        _next = next;
    26	        _logger = logger;
    27	    }
    28	
    29	    public async Task InvokeAsync(HttpContext context)
    30	    {
    31	        try
    32	        {
    33	            await _next(context);
    34	        }
    35	        catch (ApiException ex)
    36	        {
    37	            _logger.LogWarning(ex, "API exception: {Message}", ex.Message);
    38	            await WriteErrorAsync(context, (int)ex.Status, ex.Message);
    39	        }
    40	        catch (Exception ex)
    41	        {
    42	            _logger.LogError(ex, "Unhandled exception processing {Method} {Path}", context.Request.Method, context.Request.Path);
    43	            await WriteErrorAsync(context, (int)HttpStatusCode.InternalServerError, "Unexpected error occurred");
    44	        }
    45	    }

    47	    private static async Task WriteErrorAsync(HttpContext context, int status, string message)
    48	    {
    49	        if (context.Response.HasStarted) return;
    50	
    51	        context.Response.ContentType = "application/json";
    52	        context.Response.StatusCode = status;
    53	        var body = new ErrorResponse(message, status);
    54	        await context.Response.WriteAsync(JsonSerializer.Serialize(body, JsonOptions));
    55	    }
    56	}

    11	    // Java's LocalDateTime has no timezone concept at all — the default
    12	    // System.Text.Json DateTime converter writes a "+05:30"/"Z" offset
    13	    // whenever Kind is Local/Utc, which Jackson's LocalDateTimeDeserializer
    14	    // then fails to parse ("unparsed text found"). This converter always
    15	    // writes the plain wall-clock value, matching what LocalDateTime expects.
    16	    private sealed class NaiveDateTimeConverter : JsonConverter<DateTime>
    17	    {
    18	        private const string Format = "yyyy-MM-ddTHH:mm:ss.ffffff";
    19	
    20	        public override DateTime Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
    21	            => DateTime.Parse(reader.GetString()!, CultureInfo.InvariantCulture);
    22	
    23	        public override void Write(Utf8JsonWriter writer, DateTime value, JsonSerializerOptions options)
    24	            => writer.WriteStringValue(value.ToString(Format, CultureInfo.InvariantCulture));
    25	    }

    28	    // MVC in Program.cs (that only applies to ASP.NET Core's own controller
    29	    // serialization, not outgoing HttpClient calls) — without this it
    30	    // defaults to PascalCase property names, which Jackson on the Java side
    31	    // doesn't recognise (fail-on-unknown-properties is on by default there),
    32	    // so every field came through as null and the request 500'd.
    33	    private static readonly JsonSerializerOptions JavaJsonOptions = new()
    34	    {
    35	        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
    36	        Converters = { new NaiveDateTimeConverter() },
    37	    };

     6	// actual return invoice) so both price a rental the same way: pack whole
     7	// months first, then whole weeks out of what's left, then bill the
     8	// remaining days at the daily rate.
     9	public static class RentalRateCalculator
    10	{
    11	    public static double CalculateRentalAmount(CarType carType, int days)
    12	    {
    13	        var dailyRate = carType.DailyRate;
    14	        var weeklyRate = carType.WeeklyRate ?? 0.0;
    15	        var monthlyRate = carType.MonthlyRate ?? 0.0;
    16	
    17	        var remainingDays = days;
    18	        var amount = 0.0;
    19	
    20	        if (monthlyRate > 0 && remainingDays >= 30)
    21	        {
    22	            var months = remainingDays / 30;
    23	            amount += months * monthlyRate;
    24	            remainingDays %= 30;
    25	        }
    26	        if (weeklyRate > 0 && remainingDays >= 7)
    27	        {
    28	            var weeks = remainingDays / 7;
    29	            amount += weeks * weeklyRate;
    30	            remainingDays %= 7;
    31	        }
    32	        amount += remainingDays * dailyRate;
    33	
    34	        return amount;
    35	    }
    36	}

===== Security/AuthenticatedUserAccessor.cs =====
     1	using System.Security.Claims;
     2	using FlemanApi.Exceptions;
     3	using System.Net;
     4	
     5	namespace FlemanApi.Security;
     6	
     7	public record CurrentUser(long UserId, string Email, string Role);
     8	
     9	public interface IAuthenticatedUserAccessor
    10	{
    11	    bool IsAuthenticated { get; }
    12	    CurrentUser GetCurrentUser();
    13	    long GetCurrentUserId();
    14	    string GetCurrentRole();
    15	}
    16	
    17	// Mirrors com.fleman.security.AuthenticatedUser — a thin wrapper over the
    18	// current request's ClaimsPrincipal (populated by JWT bearer auth), used by
    19	// controllers/services instead of parameter-binding [ClaimsPrincipal] every time.
    20	public class AuthenticatedUserAccessor : IAuthenticatedUserAccessor
    21	{
    22	    private readonly IHttpContextAccessor _httpContextAccessor;
    23	
    24	    public AuthenticatedUserAccessor(IHttpContextAccessor httpContextAccessor)
    25	    {
    26	        _httpContextAccessor = httpContextAccessor;
    27	    }
    28	
    29	    public bool IsAuthenticated =>
    30	        _httpContextAccessor.HttpContext?.User?.Identity?.IsAuthenticated ?? false;
    31	
    32	    public CurrentUser GetCurrentUser()
    33	    {
    34	        var principal = _httpContextAccessor.HttpContext?.User
    35	            ?? throw new ApiException("Not authenticated", HttpStatusCode.Unauthorized);
    36	
    37	        var sub = principal.FindFirstValue(ClaimTypes.NameIdentifier)
    38	                  ?? principal.FindFirstValue("sub")
    39	                  ?? throw new ApiException("Not authenticated", HttpStatusCode.Unauthorized);
    40	
    41	        var email = principal.FindFirstValue("email") ?? string.Empty;
    42	        var role = principal.FindFirstValue(ClaimTypes.Role) ?? principal.FindFirstValue("role") ?? string.Empty;
    43	
    44	        return new CurrentUser(long.Parse(sub), email, role);
    45	    }
    46	
    47	    public long GetCurrentUserId() => GetCurrentUser().UserId;
    48	
    49	    public string GetCurrentRole() => GetCurrentUser().Role;
    50	}
