# CH01 - Add new vehicle categories: step-by-step instructions (Bhavana P H)

All code below was written against baseline commit `dfc61ab8` but **has not been compiled or run** (no .NET SDK was available where it was written). Expect to fix small errors when you build. Paths are relative to `WanderCar_DotNet/src/`.

## 1. What the change is

**Baseline.** A "category" is just a row in `car_types` with a name and three rates (seeded: Economy, Compact, Sedan, SUV, Luxury, Minivan). There is no category concept in code, no category-specific rule, and no endpoint to create a car type.

**Requirement.** Introduce a real vehicle category on every car type, with two new categories, **ELECTRIC** and **COMMERCIAL**, alongside **STANDARD** (all existing car types). Each category has its own rules. Staff can create a car type in any category and anyone can list car types by category.

**Category rules (business values chosen by the team - change them if you like, but record them in `requirement.md`):**

| Category | Security deposit | Rate rule |
|---|---|---|
| STANDARD | 0 | none |
| ELECTRIC | 2 x daily rate | none |
| COMMERCIAL | 5 x daily rate | weekly rate is mandatory |

**Pattern introduced.** A parameterized **Factory Method**: `VehicleCategoryRulesFactory.Create(category)` returns one of three `IVehicleCategoryRules` products. Be honest in the report that this is the parameterized variant (GoF, Factory Method, Implementation note 2), which some texts call Simple Factory; let your reviewer decide the label.

**Acceptance criteria.**
1. Existing car types keep working and are reported as `STANDARD` with deposit 0.
2. `POST /api/staff/car-types` (STAFF only) creates a car type in any of the three categories and stores the deposit computed by that category's rules.
3. A COMMERCIAL car type without a weekly rate is rejected with 400; a duplicate name is rejected with 409.
4. `GET /api/car-types/category/{category}` returns only car types of that category.
5. All 183 existing tests pass with no edits to existing test files.
6. New tests cover the three rule classes, the factory, the service and the validator.

**Deliberately out of scope** (to avoid colliding with CH02 and CH03): pricing, booking, the return invoice, the seed data. `VehicleService` and its tests are not touched.

```csharp
public class ElectricCategoryRules : IVehicleCategoryRules
```

**`CommercialCategoryRules.cs`** (ConcreteProduct)
```csharp
using FlemanApi.Models;

namespace FlemanApi.Service.Categories;

public class CommercialCategoryRules : IVehicleCategoryRules
{
    private const double DepositMultiplier = 5.0;

    public VehicleCategory Category => VehicleCategory.COMMERCIAL;
    public double CalculateSecurityDeposit(double dailyRate) => dailyRate * DepositMultiplier;

    public IReadOnlyList<string> ValidateRates(double dailyRate, double? weeklyRate, double? monthlyRate)
    {
        var errors = new List<string>();
        if (weeklyRate is null) errors.Add("weeklyRate is required for COMMERCIAL car types");
        return errors;
    }
}
```

**`VehicleCategoryRulesFactory.cs`** (Creator with the parameterized factory method)
```csharp
using System.Net;
using FlemanApi.Exceptions;
using FlemanApi.Models;

namespace FlemanApi.Service.Categories;

public interface IVehicleCategoryRulesFactory
{
    IVehicleCategoryRules Create(VehicleCategory category);
}

public class VehicleCategoryRulesFactory : IVehicleCategoryRulesFactory
{
    public IVehicleCategoryRules Create(VehicleCategory category) => category switch
    {
        VehicleCategory.STANDARD => new StandardCategoryRules(),
        VehicleCategory.ELECTRIC => new ElectricCategoryRules(),
        VehicleCategory.COMMERCIAL => new CommercialCategoryRules(),
        _ => throw new ApiException($"Unsupported vehicle category: {category}", HttpStatusCode.BadRequest),
    };
}
```

## 6. Add the service

```csharp
using FlemanApi.Models;

namespace FlemanApi.Service;

public interface ICarTypeService
{
    Task<CarTypeDTO> CreateCarTypeAsync(CreateCarTypeRequest request);
    Task<IReadOnlyList<CarTypeDTO>> GetCarTypesByCategoryAsync(VehicleCategory category);
}
```

**`FlemanApi/Service/CarTypeService.cs`**
```csharp
using System.Net;
using AutoMapper;
using FlemanApi.DTO;
using FlemanApi.Exceptions;
using FlemanApi.Models;
using FlemanApi.Repository;
using FlemanApi.Service.Categories;

namespace FlemanApi.Service;

public class CarTypeService : ICarTypeService
{
    private readonly IGenericRepository<CarType, long> _carTypes;
    private readonly IVehicleCategoryRulesFactory _rulesFactory;
    private readonly IMapper _mapper;

    public CarTypeService(
        IGenericRepository<CarType, long> carTypes,
        IVehicleCategoryRulesFactory rulesFactory,
        IMapper mapper)
    {
        _carTypes = carTypes;
        _rulesFactory = rulesFactory;
        _mapper = mapper;
    }

    public async Task<CarTypeDTO> CreateCarTypeAsync(CreateCarTypeRequest request)
    {
        var name = request.CarTypeName.Trim();
        var lower = name.ToLower();
        if (await _carTypes.ExistsAsync(c => c.CarTypeName.ToLower() == lower))
        {
            throw new ApiException($"A car type with this name already exists: {name}", HttpStatusCode.Conflict);
        }

        var rules = _rulesFactory.Create(request.Category!.Value);
        var dailyRate = request.DailyRate!.Value;

        var errors = rules.ValidateRates(dailyRate, request.WeeklyRate, request.MonthlyRate);
        if (errors.Count > 0)
        {
            throw new ApiException(string.Join(", ", errors), HttpStatusCode.BadRequest);
        }

        var carType = new CarType
        {
            CarTypeName = name,
            DailyRate = dailyRate,
            WeeklyRate = request.WeeklyRate,
            MonthlyRate = request.MonthlyRate,
            ImagePath = request.ImagePath,
            Category = rules.Category,
            SecurityDeposit = rules.CalculateSecurityDeposit(dailyRate),
        };

        await _carTypes.AddAsync(carType);
        await _carTypes.SaveChangesAsync();
        return _mapper.Map<CarTypeDTO>(carType);
    }

    public async Task<IReadOnlyList<CarTypeDTO>> GetCarTypesByCategoryAsync(VehicleCategory category)
    {
        var carTypes = await _carTypes.FindAsync(c => c.Category == category);
        return _mapper.Map<List<CarTypeDTO>>(carTypes);
    }
}
```

## 7. Add the controller and register everything

**`FlemanApi/Controllers/CarTypeController.cs`** - a new controller on purpose: `VehicleController` has `[AllowAnonymous]` on the class, and a class-level `[AllowAnonymous]` overrides `[Authorize]` on an action, so a staff-only action cannot live there.
```csharp
using FlemanApi.DTO;
using FlemanApi.Models;
using FlemanApi.Service;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace FlemanApi.Controllers;

[ApiController]
public class CarTypeController : ControllerBase
{
    private readonly ICarTypeService _carTypeService;

    public CarTypeController(ICarTypeService carTypeService)
    {
        _carTypeService = carTypeService;
    }

    [HttpGet("api/car-types/category/{category}")]
    [AllowAnonymous]
    public async Task<IActionResult> GetByCategory(VehicleCategory category) =>
        Ok(await _carTypeService.GetCarTypesByCategoryAsync(category));

    [HttpPost("api/staff/car-types")]
    [Authorize(Roles = "STAFF")]
    public async Task<IActionResult> CreateCarType([FromBody] CreateCarTypeRequest request) =>
        Ok(await _carTypeService.CreateCarTypeAsync(request));
}
```

```csharp
using FlemanApi.Exceptions;
using FlemanApi.Models;

using FluentAssertions;

namespace FlemanApi.Tests.Services;

[TestFixture]
public class VehicleCategoryRulesTests
{
    private readonly VehicleCategoryRulesFactory _factory = new();

    [Test]
    public void Create_ReturnsMatchingRulesForEachCategory()
    {
        _factory.Create(VehicleCategory.STANDARD).Should().BeOfType<StandardCategoryRules>();
        _factory.Create(VehicleCategory.ELECTRIC).Should().BeOfType<ElectricCategoryRules>();
        _factory.Create(VehicleCategory.COMMERCIAL).Should().BeOfType<CommercialCategoryRules>();
    }

    [Test]
    public void Create_EveryEnumValueIsSupportedAndReportsItsOwnCategory()
    {
        foreach (var category in Enum.GetValues<VehicleCategory>())
        {
            _factory.Create(category).Category.Should().Be(category);
        }
    }

    [Test]
    public void Create_UnknownCategory_ThrowsApiException()
    {
        var act = () => _factory.Create((VehicleCategory)999);
        act.Should().Throw<ApiException>();
    }

    [Test]
    public void Standard_HasNoDepositAndNoRateRule()
    {
        var rules = new StandardCategoryRules();
        rules.CalculateSecurityDeposit(100.0).Should().Be(0.0);
        rules.ValidateRates(100.0, null, null).Should().BeEmpty();
    }

    [Test]
    public void Electric_DepositIsTwiceTheDailyRate()
    {
        var rules = new ElectricCategoryRules();
        rules.CalculateSecurityDeposit(60.0).Should().Be(120.0);
        rules.ValidateRates(60.0, null, null).Should().BeEmpty();
    }

    [Test]
    public void Commercial_DepositIsFiveTimesTheDailyRate()
    {
        new CommercialCategoryRules().CalculateSecurityDeposit(80.0).Should().Be(400.0);
    }

    [Test]
    public void Commercial_RequiresWeeklyRate()
    {
        var rules = new CommercialCategoryRules();
        rules.ValidateRates(80.0, null, null).Should().ContainSingle();
        rules.ValidateRates(80.0, 500.0, null).Should().BeEmpty();
    }
}
```
